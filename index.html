// 防抖静默同步函数（5秒防抖） function autoSyncToGist(showTip = false) { const token = localStorage.getItem(PAGE_ID + '_github_token'); const gistId = localStorage.getItem(PAGE_ID + '_gist_id'); if (!token || !gistId) { if (showTip) showMsg('⚠️ 未配置 Gist，无法同步'); return; } // 清除之前的防抖定时器 if (syncDebounceTimer) { clearTimeout(syncDebounceTimer); } // 设置新的防抖定时器（5秒后执行） syncDebounceTimer = setTimeout(async () => { // 防止并发同步 if (isSyncing) { console.log('⏳ 已有同步进行中，跳过'); return; } isSyncing = true; // 获取上传按钮并开始旋转 const uploadBtn = document.getElementById('headerSaveToGistBtnDesktop') || document.getElementById('headerSaveToGistBtn'); setButtonLoading(uploadBtn, true); showMsg('☁️ 正在同步...'); const backupData = { version: '1.0', backupTime: new Date().toISOString(), notes: notes, trees: trees }; try { const getRes = await fetch(`https://api.github.com/gists/${gistId}`, { headers: { 'Authorization': `token ${token}` } }); if (!getRes.ok) throw new Error('获取 Gist 失败'); const gistData = await getRes.json(); const filename = Object.keys(gistData.files)[0] || 'history_backup.json'; const updateRes = await fetch(`https://api.github.com/gists/${gistId}`, { method: 'PATCH', headers: { 'Authorization': `token ${token}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ files: { [filename]: { content: JSON.stringify(backupData, null, 2) } } }) }); if (!updateRes.ok) throw new Error(`上传失败: ${updateRes.status}`); lastSyncTime = Date.now(); showMsg('✅ 同步成功'); } catch (err) { console.warn('同步失败:', err); showMsg('❌ 同步失败: ' + err.message); } finally { setButtonLoading(uploadBtn, false); isSyncing = false; syncDebounceTimer = null; } }, 5000); // 5秒防抖 }
